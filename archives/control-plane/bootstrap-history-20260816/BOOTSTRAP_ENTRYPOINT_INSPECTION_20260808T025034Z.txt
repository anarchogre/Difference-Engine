    21	def discover_tests(
    22	    module: ModuleType,

    24	    tests: list[tuple[str, Callable[[], None]]] = []
    25	
    26	    for name, value in inspect.getmembers(
    27	        module,
    28	        inspect.isfunction,

    39	
    40

    46	    for module_name in TEST_MODULES:

    49	                module_name

    52	            failed += 1

    54	                f"IMPORT_FAIL {module_name}"
    55	            )
    56	            traceback.print_exc()
    57	            continue
    58	
    59	        tests = discover_tests(module)
    60	        discovered += len(tests)
    61	
    62	        for test_name, test in tests:
    63	            try:
    64	                test()

    66	                failed += 1
    67	                print(

    69	                )

    72	                passed += 1
    73	                print(
    74	                    f"PASS {test_name}"
    75	                )
    76	
    77	    print(
    78	        f"DISCOVERED={discovered}"
    79	    )

    81	        f"PASSED={passed}"

    83	    print(
    84	        f"FAILED={failed}"
    85	    )
    86	
    87	    if discovered == 0:
    88	        return 2
    89	
    90	    if failed:

    92	
    93	    return 0
    94	
    95
